🏗️ KURULUM - ADIM ADIM

1. PROJE YAPISINI OLUŞTUR:
   mkdir -p lauth/src/main/java/com/lunex/lauth/{managers,listeners,commands}
   mkdir -p lauth/src/main/resources

2. JAVA DOSYALARINI YERLEŞT:
   ├─ 01-LauthPlugin.java → src/main/java/com/lunex/lauth/LauthPlugin.java
   ├─ 02-AuthManager.java → src/main/java/com/lunex/lauth/managers/AuthManager.java
   ├─ 03-PlayerListener.java → src/main/java/com/lunex/lauth/listeners/PlayerListener.java
   ├─ 04-LoginCommand.java → src/main/java/com/lunex/lauth/commands/LoginCommand.java
   └─ 05-RegisterCommand.java → src/main/java/com/lunex/lauth/commands/RegisterCommand.java

3. YAPIYLANDIRMA DOSYALARINI YERLEŞT:
   ├─ 06-plugin.yml → src/main/resources/plugin.yml
   ├─ 07-config.yml → src/main/resources/config.yml
   └─ 08-pom.xml → lauth/ (proje root'u)

4. DERLEME:
   cd lauth
   mvn clean package

5. KURULUM:
   cp target/lauth-1.0.jar /path/to/server/plugins/
   ./start.sh
